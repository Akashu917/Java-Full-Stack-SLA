namespace MVC_WebApp.Models
{
    public class Student
    {
        public int StudentId { get; set; }

        public string Department { get; set; }
        public int Age { get; set; }
    }
}

HomeController.cs:

using System.Collections.Generic;
using System.Web.Mvc;
using MVC_WebApp.Models;

namespace MVC_WebApp.Controllers
{
    public class HomeController : Controller
    {
        public ActionResult About()
        {
            // Model data
            List<Student> students = new List<Student>
            {
                new Student { StudentId = 1, StudentName = "Akash", Department = "CSE", Age = 22 },
                new Student { StudentId = 2, StudentName = "Dinesh", Department = "ECE", Age = 21 },
                new Student { StudentId = 3, StudentName = "Mohan", Department = "IT", Age = 23 }
            };

            // ViewBag
            ViewBag.Title = "Student Details";
            ViewBag.Message = "Welcome to the About Page";

            // ViewData
            ViewData["College"] = "SKP Engineering College";

            // TempData
            TempData["Status"] = "Student data loaded successfully.";

            return View(students);
        }
    }
}

<h3>If Else Example</h3>

@foreach (var student in Model)
{
    if (student.Age >= 22)
    {
        <p>@student.StudentName is 22 or above.</p>
    }
    else
    {
        <p>@student.StudentName is below 22.</p>
    }
}
