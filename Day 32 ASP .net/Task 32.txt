using System;

namespace StateManagementDemo
{
    public partial class Page1 : System.Web.UI.Page
    {
        protected void Page_Load(object sender, EventArgs e)
        {
        }

        // ViewState
        protected void btnViewState_Click(object sender, EventArgs e)
        {
            ViewState["Name"] = txtName.Text;

            lblMessage.Text =
                "ViewState saved: " + ViewState["Name"];
        }

        // Session State
        protected void btnSession_Click(object sender, EventArgs e)
        {
            Session["Name"] = txtName.Text;

            lblMessage.Text =
                "Session saved: " + Session["Name"];
        }

        // Application State
        protected void btnApplication_Click(object sender, EventArgs e)
        {
            Application["Name"] = txtName.Text;

            lblMessage.Text =
                "Application saved: " + Application["Name"];
        }
    }
}

using System;

namespace StateManagementDemo
{
    public partial class Page2 : System.Web.UI.Page
    {
        protected void Page_Load(object sender, EventArgs e)
        {
            // ViewState
            if (Page.PreviousPage != null)
            {
                lblViewState.Text =
                    "ViewState: ViewState belongs to Page 1.";
            }

            // Session
            if (Session["Name"] != null)
            {
                lblSession.Text =
                    "Session: " + Session["Name"];
            }
            else
            {
                lblSession.Text =
                    "Session: No value found";
            }

            // Application
            if (Application["Name"] != null)
            {
                lblApplication.Text =
                    "Application: " + Application["Name"];
            }
            else
            {
                lblApplication.Text =
                    "Application: No value found";
            }
        }
    }
}
