using System;
using System.Data;

namespace StudentGridView
{
    public partial class Student : System.Web.UI.Page
    {
        protected void Page_Load(object sender, EventArgs e)
        {
            if (!IsPostBack)
            {
                CreateStudentTable();
            }
        }

        private void CreateStudentTable()
        {
            // Create local DataTable
            DataTable dt = new DataTable();

            // Create columns
            dt.Columns.Add("StudentId", typeof(int));
            dt.Columns.Add("StudentName", typeof(string));
            dt.Columns.Add("Department", typeof(string));
            dt.Columns.Add("Age", typeof(int));
            dt.Columns.Add("City", typeof(string));

            // Add student records
            dt.Rows.Add(101, "Akash", "CSE", 22, "Chennai");
            dt.Rows.Add(102, "Arun", "IT", 21, "Coimbatore");
            dt.Rows.Add(103, "Dinesh", "ECE", 22, "Madurai");
            dt.Rows.Add(104, "Mohan", "CSE", 21, "Salem");
            dt.Rows.Add(105, "Chandru", "EEE", 22, "Vellore");

            // Bind DataTable to GridView
            GridView1.DataSource = dt;
            GridView1.DataBind();
        }
    }
}
