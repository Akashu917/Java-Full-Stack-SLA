page1.aspx.cs ------

using System;
using System.Web;

namespace DataTransferDemo
{
    public partial class Page1 : System.Web.UI.Page
    {
        protected void Page_Load(object sender, EventArgs e)
        {
        }

        // 1. Query String
        protected void btnQueryString_Click(object sender, EventArgs e)
        {
            string name = txtName.Text;

            Response.Redirect("Page2.aspx?name=" + Server.UrlEncode(name));
        }

        // 2. Cookies
        protected void btnCookie_Click(object sender, EventArgs e)
        {
            HttpCookie cookie = new HttpCookie("UserName");

            cookie.Value = txtName.Text;

            Response.Cookies.Add(cookie);

            Response.Redirect("Page2.aspx");
        }

        // 3. Response.Redirect
        protected void btnRedirect_Click(object sender, EventArgs e)
        {
            Response.Redirect("Page2.aspx");
        }

        // 4. Server.Transfer
        protected void btnTransfer_Click(object sender, EventArgs e)
        {
            Server.Transfer("Page2.aspx");
        }

        // 5. Server.Execute
        protected void btnExecute_Click(object sender, EventArgs e)
        {
            Server.Execute("Page2.aspx");
        }
    }
}

using System;
using System.Web;

namespace DataTransferDemo
{
    public partial class Page2 : System.Web.UI.Page
    {
        protected void Page_Load(object sender, EventArgs e)
        {
            // Query String
            string name = Request.QueryString["name"];

            if (!string.IsNullOrEmpty(name))
            {
                lblQueryString.Text ="Query String: " + Server.HtmlEncode(name);
            }
            else
            {
                lblQueryString.Text ="Query String: No data";
            }

            // Cookie
            HttpCookie cookie = Request.Cookies["UserName"];

            if (cookie != null)
            {
                lblCookie.Text ="Cookie: " + Server.HtmlEncode(cookie.Value);
            }
            else
            {
                lblCookie.Text ="Cookie: No data";
            }

            lblMessage.Text = "You are now on Page 2.";
        }
    }
}
