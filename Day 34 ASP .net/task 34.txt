2.Example for Cache.Insert?

using System;

namespace CacheDemo
{
    public partial class CacheExample : System.Web.UI.Page
    {
        protected void Page_Load(object sender, EventArgs e)
        {
            if (Cache["Message"] == null)
            {
                string message = "Hello, this data is stored in Cache!";

                Cache.Insert("Message", message);
            }

            lblMessage.Text = Cache["Message"].ToString();
        }
    }
}
