2.Write a C# program that takes a sentence as input and counts the frequency of each word using a Dictionary<string, int>. The program should then display the word frequencies.
using System;
using System.Collections.Generic;

class WordFrequency
{
    public void Dictionary()
    {
        Console.Write("Enter a Sentence: ");
        string sentence = Console.ReadLine();
        
        string[] words = sentence.ToLower().Split(' ');
        
        Dictionary<string,int> wordCount = new Dictionary<string,int>();
        
        foreach(string word in words)
        {
            if(wordCount.ContainsKey(word))
            {
                wordCount[word]++;
            }
            else
            {
                wordCount.Add(word,1);
            }
        }
        Console.WriteLine("\nWord Frequencies:");
        
        foreach (KeyValuePair<string,int> item in wordCount)
        {
            Console.WriteLine(item.Key + " : " + item.Value);
        }
    }
}

class Program
{
    static void Main()
    {
        WordFrequency w = new WordFrequency();
        w.Dictionary();
    }
}

2.write the same program without dictionary.

using System;

class WordFrequency
{
    public void WithoutDictionaryDemo()
    {
        Console.Write("enter a sentence: ");
        string sentence = Console.ReadLine();

        string[] words = sentence.ToLower().Split(' ');
        bool[] visited = new bool[words.Length];
        Console.WriteLine("\nWord Frequencies:");

        for (int i = 0; i < words.Length; i++)
        {
            if (visited[i])
            {
                continue;
            }
            
            int count = 1;
            for (int j = i + 1; j < words.Length; j++)
            {
                if (words[i] == words[j])
                {
                    count++;
                    visited[j] = true;
                }
            }

            Console.WriteLine(words[i] + " : " + count);
        }
    }
}

class Program
{
    static void Main(string[] args)
    {
        WordFrequency wf = new WordFrequency();
        wf.WithoutDictionaryDemo();
        
    }
}
