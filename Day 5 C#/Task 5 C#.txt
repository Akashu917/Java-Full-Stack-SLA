1. Count Duplicate Elements in an Array?

using System;

class Program
{
    static void Main()
    {
        Console.Write("Input the number of elements: ");
        int n = Convert.ToInt32(Console.ReadLine());

        int[] arr = new int[n];

        Console.WriteLine("Input elements:");
        for (int i = 0; i < n; i++)
        {
            Console.Write("Element - " + i + " : ");
            arr[i] = Convert.ToInt32(Console.ReadLine());
        }

        int count = 0;

        for (int i = 0; i < n; i++)
        {
            bool isDuplicate = false;

            for (int j = 0; j < i; j++)
            {
                if (arr[i] == arr[j])
                {
                    isDuplicate = true;
                    break;
                }
            }

            if (isDuplicate)
                count++;
        }

        Console.WriteLine("Total number of duplicate elements found in the array is : " + count);
    }
}

2.Merge Two Sorted Arrays in Ascending Order?

using System;

class Program
{
    static void Main()
    {
        Console.Write("Input size of first array: ");
        int n = Convert.ToInt32(Console.ReadLine());

        int[] arr1 = new int[n];
        int[] arr2 = new int[n];
        int[] merge = new int[n * 2];

        Console.WriteLine("Enter first array:");
        for (int i = 0; i < n; i++)
        {
            arr1[i] = Convert.ToInt32(Console.ReadLine());
        }

        Console.WriteLine("Enter second array:");
        for (int i = 0; i < n; i++)
        {
            arr2[i] = Convert.ToInt32(Console.ReadLine());
        }

        for (int i = 0; i < n; i++)
            merge[i] = arr1[i];

        for (int i = 0; i < n; i++)
            merge[n + i] = arr2[i];

        // Sorting
        for (int i = 0; i < merge.Length - 1; i++)
        {
            for (int j = i + 1; j < merge.Length; j++)
            {
                if (merge[i] > merge[j])
                {
                    int temp = merge[i];
                    merge[i] = merge[j];
                    merge[j] = temp;
                }
            }
        }

        Console.WriteLine("Merged array in ascending order:");

        foreach (int num in merge)
        {
            Console.Write(num+" ");
        }
    }
}



3.Print Specific Patterns (Alphabet Pyramid & Number Triangles)?

using System;

class Program
{
     // Pattern 1: A, ABA, ABCBA, ABCDCBA
    static void PrintAlphabetPyramid()
    {
        Console.Write("Enter a number :");
        int rows = Convert.ToInt32(Console.ReadLine());
        Console.WriteLine("--- Pattern 1: Alphabet Pyramid ---");
        for (int i = 1; i <= rows; i++)
        {
            // Ascending alphabets
            for (int j = 0; j < i; j++)
            {
                Console.Write((char)('A' + j));
            }
            // Descending alphabets
            for (int j = i - 2; j >= 0; j--)
            {
                Console.Write((char)('A' + j));
            }
            Console.WriteLine();
        }
    }

    // Pattern 2: 1, 121, 12321, 1234321, 123454321
    static void PrintNumberPyramid()
    {
        Console.Write("Enter a number :");
        int rows = Convert.ToInt32(Console.ReadLine());
        
        Console.WriteLine("--- Pattern 2: Palindromic Number Pyramid ---");
        for (int i = 1; i <= rows; i++)
        {
            // Ascending numbers
            for (int j = 1; j <= i; j++)
            {
                Console.Write(j);
            }
            // Descending numbers
            for (int j = i - 1; j >= 1; j--)
            {
                Console.Write(j);
            }
            Console.WriteLine();
        }
    }

    // Pattern 3: 1, 22, 333, 4444, 55555
    static void PrintRepeatedNumberTriangle()
    {
        Console.Write("Enter a number :");
        int rows = Convert.ToInt32(Console.ReadLine());
        
        Console.WriteLine("--- Pattern 3: Repeated Number Triangle ---");
        for (int i = 1; i <= rows; i++)
        {
            for (int j = 1; j <= i; j++)
            {
                Console.Write(i);
            }
            Console.WriteLine();
        }
    }
    static void Main()
    {
        
        PrintAlphabetPyramid();
        PrintNumberPyramid();
        PrintRepeatedNumberTriangle();
    }
}
