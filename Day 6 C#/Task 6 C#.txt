1.Create a student class with required details like ID,name,DOB,course,contact,add methods for calculating age?

using System;

internal class Student
{
    public int Id { get; set; }
    public string Name { get; set; }
    public DateTime DOB { get; set; }
    public string Course { get; set; }
    public string Contact { get; set; }
}

class Details : Student
{
    public void StudentDetails()
    {
        Id = 101;
        Name = "Akash";
        DOB = new DateTime(2004, 12, 28);
        Course = ".NET";
        Contact = "9360305917";
    }

    public int CalculateAge()
    {
        int age = DateTime.Now.Year - DOB.Year;

        if (DateTime.Now < DOB.AddYears(age))
        {
            age--;
        }

        return age;
    }

    public void Display()
    {
        Console.WriteLine("Student ID: " + Id);
        Console.WriteLine("Name: " + Name);
        Console.WriteLine("DOB: " + DOB.ToShortDateString());
        Console.WriteLine("Course: " + Course);
        Console.WriteLine("Contact: " + Contact);
        Console.WriteLine("Age: " + CalculateAge());
    }
}

using System;

class Program
{
    public static void Main(string[] args)
    {
        Details student = new Details();

        student.StudentDetails();

        Console.ReadLine();
    }
}

3.Write a c# program to check palindrome number?

using System;
using task;

class Program
{
    public static void Main(string[] args)
    {
        Console.Write("Enter a Number :");
        int num = Convert.ToInt32(Console.ReadLine());
        int orginal = num;
        int reverse = 0;

        while (num > 0)
        {
            int digit = num % 10;
            reverse = reverse * 10 + digit;
            num = num / 10;

        }

        if (orginal == reverse)
        {
            Console.WriteLine("Palindrome Number");
        }
        else
        {
            Console.WriteLine("Not a Palaindrome Number");
        }

    }
}
