1.Write a C# function that takes an array of Integers as input and returns a new array where each element is the product of all the numbers in the original array except the one at that index. For example: Task: Input: [1,2,3,4] Output: (24, 12, 6,8) ?


using System;
class Program
{
    public static int[] Product(int[] nums)
    {
        // handle edge cases
        if (nums == null || nums.Length == 0)
        {
            return new int[0];
        }
        if(nums.Length == 1)
        {
            return new int[] {1};
        }
        int n = nums.Length;
        int[] result = new int[n];
        
        //left products
        result[0] = 1;
        for(int i = 1; i < n; i++)
        {
            result[i] = result[i - 1]* nums[i - 1];
        }
        
        //right products
        int rightProduct = 1;
        for (int i = n - 1;i>= 0; i-- )
        {
            result[i] *= rightProduct;
            rightProduct *= nums[i];
        }
        return result;
    }
    
    static void Main()
    {
        int[] nums = {1,2,3,4};
        int[] result = Product(nums);
        
        Console.Write("output:"+"[" + string.Join(",",result)+"]");
    }
}
